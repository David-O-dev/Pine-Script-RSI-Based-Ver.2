//@version=6
indicator("RSI Based", overlay = true, precision = 5)


EMA10 = request.security(syminfo.tickerid, timeframe = "3", expression = ta.ema(close, 200))

//200 EMA
bool Above200 = false
bool Below200 = false
if close > EMA10
    Above200 := true
    Below200 := false

else if close < EMA10
    Above200 := false
    Below200 := true




//RSI
//User Inputs
UseRSI = input.bool(defval = true, title = "Use RSI", group = "RSI", tooltip = "RSI 55이상/45이하이고 정배열일 시 롱/숏 표시")
RSITimeFrame1 = input.string(defval = "3", title = "RSI Time Frame1", group = "RSI")
RSITimeFrame2 = input.string(defval = "5", title = "RSI Time Frame2", group = "RSI")
RSITimeFrame3 = input.string(defval = "10", title = "RSI Time Frame3", group = "RSI")

//RSI Length
rsi1Length = input.int(defval = 7, title = "RSI 1 Line", group = "RSI")
rsi2Length = input.int(defval = 9, title = "RSI 2 Line", group = "RSI")
rsi3Length = input.int(defval = 12, title = "RSI 3 Line", group = "RSI")

//RSI Time Frame 1
RSI1 = request.security(syminfo.tickerid, timeframe = RSITimeFrame1, expression = ta.rsi(close, rsi1Length))
RSI2 = request.security(syminfo.tickerid, timeframe = RSITimeFrame1, expression = ta.rsi(close, rsi2Length))
RSI3 = request.security(syminfo.tickerid, timeframe = RSITimeFrame1, expression = ta.rsi(close, rsi3Length))

//RSI Time Frame 2
RSI4 = request.security(syminfo.tickerid, timeframe = RSITimeFrame2, expression = ta.rsi(close, rsi1Length))
RSI5 = request.security(syminfo.tickerid, timeframe = RSITimeFrame2, expression = ta.rsi(close, rsi2Length))
RSI6 = request.security(syminfo.tickerid, timeframe = RSITimeFrame2, expression = ta.rsi(close, rsi3Length))

//RSI Time Frame 3
RSI7 = request.security(syminfo.tickerid, timeframe = RSITimeFrame3, expression = ta.rsi(close, rsi1Length))
RSI8 = request.security(syminfo.tickerid, timeframe = RSITimeFrame3, expression = ta.rsi(close, rsi2Length))
RSI9 = request.security(syminfo.tickerid, timeframe = RSITimeFrame3, expression = ta.rsi(close, rsi3Length))

//Below 45 or Above 55 and In Order (1-3)
bool To3RSILong = false
bool To3RSIShort = false
bool To3RSICrossUp = false
bool To3RSICrossDown = false
if UseRSI
    if RSI1 > 55 and RSI2 > 55 and RSI3 > 55
        if RSI1 > RSI2 and RSI2 > RSI3
            To3RSILong := true
            To3RSIShort := false


    else if RSI1 < 45 and RSI2 < 45 and RSI3 < 45
        if RSI1 < RSI2 and RSI2 < RSI3
            To3RSILong := false
            To3RSIShort := true

    else
        To3RSILong := false
        To3RSIShort := false

    if RSI3[0] > 55 and RSI3[1] <= 55
        To3RSICrossUp := true
        To3RSICrossDown := false
        
    else if RSI3[0] < 45 and RSI3[1] >= 45
        To3RSICrossUp := false
        To3RSICrossDown := true

//Below 45 or Above 55 and In Order (4-6)
bool To6RSILong = false
bool To6RSIShort = false
bool To6RSICrossUp = false
bool To6RSICrossDown = false
if UseRSI
    if RSI4 > 55 and RSI5 > 55
        if RSI4 > RSI5 and RSI5 > RSI6
            To6RSILong := true
            To6RSIShort := false


    else if RSI4 < 45 and RSI5 < 45
        if RSI4 < RSI5 and RSI5 < RSI6
            To6RSILong := false
            To6RSIShort := true

    else
        To6RSILong := false
        To6RSIShort := false

    if RSI5[0] > 55 and RSI5[1] <= 55
        To6RSICrossUp := true
        To6RSICrossDown := false
            
    else if RSI5[0] < 45 and RSI5[1] >= 45
        To6RSICrossUp := false
        To6RSICrossDown := true


//Below 45 or Above 55 and In Order (7-9)
bool To9RSILong = false
bool To9RSIShort = false
bool To9RSICrossUp = false
bool To9RSICrossDown = false
if UseRSI
    if RSI7 > 55
        if RSI7 > RSI8 and RSI8 > RSI9
            To9RSILong := true
            To9RSIShort := false


    else if RSI7 < 45
        if RSI7 < RSI8 and RSI8 < RSI9
            To9RSILong := false
            To9RSIShort := true

    else
        To9RSILong := false
        To9RSIShort := false

    if RSI7[0] > 55 and RSI7[1] <= 55
        To9RSICrossUp := true
        To9RSICrossDown := false
            
    else if RSI7[0] < 45 and RSI7[1] >= 45
        To9RSICrossUp := false
        To9RSICrossDown := true






//MACD
//User input
UseMACD = input.bool(defval = true, title = "Use MACD", group = "MACD", tooltip = "MACD 영선 통과 시 롱/숏 표시")
MACDTimeFrame = input.string(defval = "1", title = "MACD Time Frame", group = "MACD")

//Getting MACD Values
[macdLine, signal, histogram] = request.security(syminfo.tickerid, timeframe = MACDTimeFrame, expression = ta.macd(close, 12, 26, 9))
MACDZeroLine = 0

//Detect MACD Cross
bool macdLong = false
bool macdShort = false
if UseMACD
    if macdLine[0] > MACDZeroLine
        macdLong := true
        macdShort := false

    else if macdLine[0] < MACDZeroLine
        macdLong := false
        macdShort := true

    else
        macdLong := false
        macdShort := false








//Ichimoku
//User Inputs
UseIchimoku = input.bool(defval = true, title = "Use Ichimoku", group = "Ichimoku", tooltip = "전환선이 위(롱)/아래(숏)으로 꺾였을 시 롱/숏 표시")

//Calculations
conversionPeriods = input.int(9, minval=1, title="Conversion Line Length", group = "Ichimoku")
donchian(len) => math.avg(ta.lowest(len), ta.highest(len))
conversionLine = donchian(conversionPeriods)

//Conversion Line Facing Up or Down
bool ConversionLineFacingUp = false
bool ConversionLineFacingDown = false
if UseIchimoku
    if conversionLine[1] < conversionLine[0]
        ConversionLineFacingUp := true
        ConversionLineFacingDown := false

    else if conversionLine[1] > conversionLine[0]
        ConversionLineFacingUp := false
        ConversionLineFacingDown := true

    else
        ConversionLineFacingUp := false
        ConversionLineFacingDown := false




if Above200 and (UseIchimoku ? ConversionLineFacingUp : true) and (UseMACD ? macdLong : true) and (UseRSI ? To3RSILong : true) and (UseRSI ? To6RSILong : true) and (UseRSI ? To9RSILong : true) and timeframe.period == "3"
    if To3RSICrossUp or To6RSICrossUp or To9RSICrossUp
        label.new(bar_index, high, "Long", color = color.rgb(94, 228, 77), size = size.small)

else if Below200 and (UseIchimoku ? ConversionLineFacingDown : true) and (UseMACD ? macdShort : true) and (UseRSI ? To3RSIShort : true) and (UseRSI ? To6RSIShort : true) and (UseRSI ? To9RSIShort : true) and timeframe.period == "3"
    if To3RSICrossDown or To6RSICrossDown or To9RSICrossDown
        label.new(bar_index, high, "Short", color = color.rgb(243, 170, 33), size = size.small)
